cmake_minimum_required(VERSION 3.22)
project(electromux_node LANGUAGES CXX)
if(NOT ANDROID_ABI STREQUAL "arm64-v8a")
  message(FATAL_ERROR "Independent proof currently supports ARM64 only")
endif()
add_library(node SHARED IMPORTED)
file(SHA256 "${NODE_SDK_ROOT}/bin/${ANDROID_ABI}/libnode.so" NODE_LIBRARY_SHA256)
if(NOT NODE_LIBRARY_SHA256 STREQUAL "955b308b1dfdf7662e8fe5ee4eb8c0c7d0a313f2d306387f2307529993c4bc32")
  message(FATAL_ERROR "Node library does not match pinned v24.21.0-0 full ARM64 artifact")
endif()
set_target_properties(node PROPERTIES IMPORTED_LOCATION "${NODE_SDK_ROOT}/bin/${ANDROID_ABI}/libnode.so")
add_library(electromux_node SHARED embedded_node.cpp)
target_include_directories(electromux_node SYSTEM PRIVATE "${NODE_SDK_ROOT}/include/node")
target_link_libraries(electromux_node PRIVATE node log)
target_link_options(electromux_node PRIVATE "-Wl,-z,max-page-size=16384")
